/*

        SKINCARE E-COMMERCE SALES ANALYSIS USING SQL

PROJECT OBJECTIVE:

Analyze customer behavior, sales performance,
product performance, returns and customer satisfaction
using SQL.


TABLES USED:

1. customers
2. orders
3. order_items
4. products
5. returns
6. reviews


SQL CONCEPTS:

SELECT | WHERE | GROUP BY | HAVING
JOIN | CASE WHEN | SUBQUERY
CTE | WINDOW FUNCTIONS
DATE ANALYSIS | DATA QUALITY CHECKS

====================================================
*/


/*
====================================================
SECTION 1: CUSTOMER ANALYSIS
====================================================
*/


-- 1. Find customers who generated the highest revenue

SELECT
    customer_id,
    SUM(final_amount) AS total_revenue
FROM orders
GROUP BY customer_id
ORDER BY total_revenue DESC
LIMIT 10;


-- 2. Find repeat customers who purchased more than once

SELECT
    customer_id,
    COUNT(order_id) AS total_orders
FROM orders
GROUP BY customer_id
HAVING COUNT(order_id) > 1
ORDER BY total_orders DESC;


-- 3. Customer value segmentation based on spending

SELECT
    customer_id,
    SUM(final_amount) AS spending,

    CASE
        WHEN SUM(final_amount) >= 50000 THEN 'High Value'
        WHEN SUM(final_amount) >= 20000 THEN 'Medium Value'
        ELSE 'Low Value'
    END AS customer_segment

FROM orders
GROUP BY customer_id;


-- 4. Which acquisition channel brings the most customers?

SELECT
    acquisition_channel,
    COUNT(customer_id) AS customers
FROM customers
GROUP BY acquisition_channel
ORDER BY customers DESC;


-- 5. Revenue generated by each customer city

SELECT
    c.city,
    SUM(o.final_amount) AS revenue
FROM customers c
JOIN orders o
ON c.customer_id = o.customer_id
GROUP BY c.city
ORDER BY revenue DESC;


/*
====================================================
SECTION 2: SALES PERFORMANCE ANALYSIS
====================================================
*/


-- 6. Overall sales performance

SELECT
    COUNT(order_id) AS total_orders,
    SUM(final_amount) AS revenue,
    AVG(final_amount) AS average_order_value
FROM orders;


-- 7. Monthly revenue trend

SELECT
    MONTH(STR_TO_DATE(order_date,'%Y-%m-%d')) AS month,
    SUM(final_amount) AS revenue
FROM orders
GROUP BY month
ORDER BY month;


-- 8. Sales performance by order status

SELECT
    order_status,
    COUNT(order_id) AS orders,
    SUM(final_amount) AS revenue
FROM orders
GROUP BY order_status
ORDER BY revenue DESC;


-- 9. Which payment method generates highest revenue?

SELECT
    payment_method,
    SUM(final_amount) AS revenue
FROM orders
GROUP BY payment_method
ORDER BY revenue DESC;


-- 10. Which sales channel performs best?

SELECT
    sales_channel,
    COUNT(order_id) AS orders,
    SUM(final_amount) AS revenue
FROM orders
GROUP BY sales_channel
ORDER BY revenue DESC;


/*
====================================================
SECTION 2: SALES PERFORMANCE ANALYSIS
====================================================
*/


-- 11. Top 10 highest value orders

SELECT
    order_id,
    final_amount
FROM orders
ORDER BY final_amount DESC
LIMIT 10;


-- 12. Month wise order growth analysis

SELECT
    MONTH(STR_TO_DATE(order_date,'%Y-%m-%d')) AS month,
    COUNT(order_id) AS orders
FROM orders
GROUP BY month
ORDER BY month;


/*
====================================================
SECTION 3: PRODUCT PERFORMANCE ANALYSIS
====================================================
*/


-- 13. Top selling products by quantity

SELECT
    product_id,
    SUM(quantity) AS units_sold
FROM order_items
GROUP BY product_id
ORDER BY units_sold DESC
LIMIT 10;


-- 14. Highest revenue generating products

SELECT
    product_id,
    SUM(item_total) AS revenue
FROM order_items
GROUP BY product_id
ORDER BY revenue DESC
LIMIT 10;


-- 15. Category wise revenue analysis

SELECT
    p.category,
    SUM(oi.item_total) AS revenue
FROM products p
JOIN order_items oi
ON p.product_id = oi.product_id
GROUP BY p.category
ORDER BY revenue DESC;


-- 16. Product profit analysis

SELECT
    product_name,
    category,
    (mrp - cost_price) AS profit
FROM products
ORDER BY profit DESC;


-- 17. Products with high stock risk

SELECT
    product_name,
    stock_qty
FROM products
WHERE stock_qty < 50
ORDER BY stock_qty;


-- 18. Products never purchased

SELECT
    product_name
FROM products
WHERE product_id NOT IN
(
    SELECT product_id
    FROM order_items
);


/*
====================================================
SECTION 4: ADVANCED SQL ANALYSIS
====================================================
*/


-- 19. Rank products based on sales

SELECT
    product_id,
    SUM(quantity) AS total_sales,

    RANK() OVER
    (
        ORDER BY SUM(quantity) DESC
    ) AS product_rank

FROM order_items
GROUP BY product_id;


-- 20. Top customers using ranking

SELECT
    customer_id,
    SUM(final_amount) AS spending,

    DENSE_RANK() OVER
    (
        ORDER BY SUM(final_amount) DESC
    ) AS customer_rank

FROM orders
GROUP BY customer_id;


/*
====================================================
SECTION 5: RETURN ANALYSIS
====================================================
*/

-- 21. Calculate overall return rate

SELECT
    COUNT(return_id) * 100.0 /
    (SELECT COUNT(*) FROM orders) AS return_rate_percentage
FROM returns;


-- 22. Most common return reasons

SELECT
    return_reason,
    COUNT(return_id) AS total_returns
FROM returns
GROUP BY return_reason
ORDER BY total_returns DESC;


-- 23. Products with highest returns

SELECT
    product_id,
    COUNT(return_id) AS return_count
FROM returns
GROUP BY product_id
ORDER BY return_count DESC
LIMIT 10;


-- 24. Refund status analysis

SELECT
    refund_status,
    COUNT(return_id) AS total_cases
FROM returns
GROUP BY refund_status
ORDER BY total_cases DESC;


-- 25. High sales but high return products

SELECT
    oi.product_id,
    SUM(oi.quantity) AS units_sold,
    COUNT(r.return_id) AS total_returns
FROM order_items oi
JOIN returns r
ON oi.product_id = r.product_id
GROUP BY oi.product_id
HAVING COUNT(r.return_id) > 5
ORDER BY total_returns DESC;



/*
====================================================
SECTION 6: REVIEW ANALYSIS
====================================================
*/

-- 26. Average product rating

SELECT
    product_id,
    AVG(rating) AS avg_rating
FROM reviews
GROUP BY product_id
ORDER BY avg_rating DESC;


-- 27. Low rated products (quality issues)

SELECT
    product_id,
    AVG(rating) AS avg_rating
FROM reviews
GROUP BY product_id
HAVING AVG(rating) < 3
ORDER BY avg_rating ASC;


-- 28. Most reviewed products

SELECT
    product_id,
    COUNT(review_id) AS total_reviews
FROM reviews
GROUP BY product_id
ORDER BY total_reviews DESC
LIMIT 10;


-- 29. Sales vs Rating comparison

SELECT
    p.product_name,
    SUM(oi.item_total) AS total_sales,
    AVG(r.rating) AS avg_rating
FROM products p
JOIN order_items oi ON p.product_id = oi.product_id
JOIN reviews r ON p.product_id = r.product_id
GROUP BY p.product_name;



/*
====================================================
SECTION 7: ADVANCED BUSINESS ANALYSIS
====================================================
*/

-- 30. Category wise profit

SELECT
    p.category,
    SUM(p.mrp - p.cost_price) AS total_profit
FROM products p
GROUP BY p.category;


-- 31. Customer lifetime value

SELECT
    customer_id,
    SUM(final_amount) AS lifetime_value
FROM orders
GROUP BY customer_id
ORDER BY lifetime_value DESC;


-- 32. Customers with no orders (data quality check)

SELECT
    c.customer_id,
    c.customer_name
FROM customers c
LEFT JOIN orders o
ON c.customer_id = o.customer_id
WHERE o.order_id IS NULL;


-- 33. Running total sales

SELECT
    order_date,
    final_amount,
    SUM(final_amount) OVER (ORDER BY order_date) AS running_sales
FROM orders;


-- 34. Month-over-month comparison (LAG)

SELECT
    month,
    revenue,
    LAG(revenue) OVER (ORDER BY month) AS prev_month_revenue
FROM (
    SELECT
        MONTH(STR_TO_DATE(order_date,'%Y-%m-%d')) AS month,
        SUM(final_amount) AS revenue
    FROM orders
    GROUP BY month
) t;


-- 35. Duplicate orders check

SELECT
    order_id,
    COUNT(*) AS duplicate_count
FROM orders
GROUP BY order_id
HAVING COUNT(*) > 1;


-- 36. Missing customer data handling

SELECT
    customer_id,
    COALESCE(customer_name, 'Unknown') AS customer_name
FROM customers;


-- 37. Product popularity (reviews + ratings)

SELECT
    p.product_name,
    COUNT(r.review_id) AS total_reviews,
    AVG(r.rating) AS avg_rating
FROM products p
JOIN reviews r ON p.product_id = r.product_id
GROUP BY p.product_name
ORDER BY total_reviews DESC;


-- 38. Customer engagement analysis

SELECT
    customer_id,
    COUNT(order_id) AS total_orders,
    SUM(final_amount) AS total_spent
FROM orders
GROUP BY customer_id;


-- 39. Category wise sales contribution

SELECT
    p.category,
    SUM(oi.item_total) AS category_sales
FROM products p
JOIN order_items oi ON p.product_id = oi.product_id
GROUP BY p.category
ORDER BY category_sales DESC;


-- 40. Final business summary report

SELECT
    COUNT(DISTINCT o.customer_id) AS total_customers,
    COUNT(o.order_id) AS total_orders,
    SUM(o.final_amount) AS total_revenue,
    AVG(o.final_amount) AS avg_order_value
FROM orders o;
